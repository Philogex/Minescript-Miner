cmake_minimum_required(VERSION 3.20)
project(MinecraftMiner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS native/src/aim/*.cpp
    native/src/catalog/*.cpp native/src/geometry/*.cpp native/src/scanner/*.cpp)
add_library(miner_core STATIC native/src/api.cpp ${CORE_SOURCES})
target_include_directories(miner_core PUBLIC native/include third_party/boost)
if(MSVC)
    target_compile_options(miner_core PRIVATE /bigobj)
endif()
option(MINER_BUILD_JNI "Build the Java binding" ON)
if(MINER_BUILD_JNI)
    find_package(JNI REQUIRED)
    add_library(miner_jni SHARED native/src/jni.cpp)
    target_include_directories(miner_jni PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(miner_jni PRIVATE miner_core)
    # Keep the DLL beside the Linux library path, including multi-config builds.
    set_target_properties(miner_jni PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<0:>"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<0:>")
endif()

include(CTest)
if(BUILD_TESTING)
    foreach(name geometry projection constraint_region constraint_subtraction branch_bound aim_target_region catalog)
        add_executable(${name}_test native/tests/${name}_test.cpp)
        target_link_libraries(${name}_test PRIVATE miner_core)
        add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
    add_executable(geometry_regression_test native/tests/geometry_regression_test.cpp)
    target_link_libraries(geometry_regression_test PRIVATE miner_core)
    foreach(case robust_orientation near_plane reach thin_sliver target_corner
            float_camera_edge_clearance adjacent_full_cube_occlusion)
        add_test(NAME regression_${case} COMMAND geometry_regression_test ${case})
    endforeach()
    add_executable(scan_pipeline_test native/tests/scan_pipeline_test.cpp native/tests/scan_fixture.cpp)
    target_link_libraries(scan_pipeline_test PRIVATE miner_core)
    file(GLOB fixtures CONFIGURE_DEPENDS native/tests/fixtures/*.scan)
    foreach(fixture ${fixtures})
        get_filename_component(name ${fixture} NAME_WE)
        add_test(NAME scan_${name} COMMAND scan_pipeline_test ${fixture})
    endforeach()
    add_executable(random_scan_pipeline_test native/tests/random_scan_pipeline_test.cpp)
    target_link_libraries(random_scan_pipeline_test PRIVATE miner_core)
    add_test(NAME random_scan COMMAND random_scan_pipeline_test 0x1234 3 0.25 5)
    get_property(targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(target ${targets})
        if(target MATCHES "_test$")
            # Existing native tests use assert; keep them active in Release.
            if(MSVC)
                target_compile_options(${target} PRIVATE /UNDEBUG)
            else()
                target_compile_options(${target} PRIVATE -UNDEBUG)
            endif()
        endif()
    endforeach()
endif()
